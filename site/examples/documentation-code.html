Шесть вариантов примера, синхронный выбор языка и одна видимая панель в каждом блоке. Это образец оформления, а не контракт API.

Пример 01

Выберите любой язык. Второй блок переключится вместе с ним; кнопка Copy копирует только свой фрагмент, без названия вкладки.

cURL / DEMO
# Illustrative URL: supply your own API_BASE
curl --fail "$API_BASE/v1/documents/demo"
Python / DEMO
document = {"name": "demo-document"}
print(document["name"])
Go / DEMO
package main

import "fmt"

func main() { fmt.Println("demo-document") }
TypeScript / DEMO
const documentName: string = "demo-document";
console.log(documentName);
Vue / DEMO
<template>
  <p>demo-document</p>
</template>
React / DEMO
export default function Document() {
  return <p>demo-document</p>;
}

Пример 02

Вкладки доступны через Tab, стрелки, Home и End. После подтверждённого копирования появляются зелёная галочка и уведомление.

cURL / DEMO
# Illustrative URL: supply your own API_BASE
curl --fail "$API_BASE/v1/documents/demo/status"
Python / DEMO
status = "demo"
print(status)
Go / DEMO
package main

import "fmt"

func main() { fmt.Println("status: demo") }
TypeScript / DEMO
const status: string = "demo";
console.log(status);
Vue / DEMO
<template>
  <output aria-live="polite">status: demo</output>
</template>
React / DEMO
export default function Status() {
  return <output aria-live="polite">status: demo</output>;
}
