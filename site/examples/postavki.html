    Компонент не знает адреса API. Приложение передаёт обработчик onSubmit(files); при ошибке выбранные страницы сохраняются.

    // submitFiles — обработчик вашего приложения, не API дизайн-системы.
const upload = SacraUpload.mount(document.querySelector('.s-upload'), {
  maxFiles: 10,
  maxBytes: 10 * 1024 * 1024,
  onSubmit: files => submitFiles(files)
});
// При удалении экрана освободите Object URL и обработчики:
upload.destroy();
